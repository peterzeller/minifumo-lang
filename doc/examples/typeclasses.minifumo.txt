

fun showIntLocal() Show[Int]
    Show((value Int) -> "41")

fun showInt2Local() Show[Int]
    Show((value Int) -> "42")

fun printValue[T](value T, s Show[T]) unit
    println(value, s)

fun main()
    printValue("Hello World", showString())
    printValue(42, showInt2Local())


// Output:
// Hello World
// 42
