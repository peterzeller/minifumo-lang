

// Shows an Int as "41".
fun showIntLocalFn(value Int) String
    "41"

fun showIntLocal() Show[Int]
    Show(showIntLocalFn)

// Shows an Int as "42".
fun showInt2LocalFn(value Int) String
    "42"

fun showInt2Local() Show[Int]
    Show(showInt2LocalFn)

// Prints a value using an explicit Show dictionary.
fun printValue[T](value T, s Show[T]) unit
    println(value, s)

fun main()
    printValue("Hello World", showString())
    printValue(42, showInt2Local())


// Output:
// Hello World
// 42
