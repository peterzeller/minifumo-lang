// Proves associativity of natAdd by induction on the first argument.
export fun add_assoc(x: Nat, y: Nat, z: Nat): Eq[Nat, natAdd(x, natAdd(y, z)), natAdd(natAdd(x, y), z)]
    match x
        case Zero
            refl
        case Suc(n)
            let h1: Eq[Nat, natAdd(Suc(natAdd(n, y)), z), Suc(natAdd(natAdd(n, y), z))] = refl
            let ih: Eq[Nat, natAdd(natAdd(n, y), z), natAdd(n, natAdd(y, z))] = add_assoc(n, y, z)
            let h2: Eq[Nat, natAdd(Suc(natAdd(n, y)), z), Suc(natAdd(n, natAdd(y, z)))] = subst(ih, h1)
            h2
